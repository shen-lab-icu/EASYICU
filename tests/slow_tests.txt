#
# 为什么用清单而不是在 87 个测试函数上加装饰器:
#   慢测试在文件内是分散的 (test_pipeline.py 的 284 个测试里只有 14 个慢),
#   模块级 pytestmark 会把不慢的一起误伤; 逐个加装饰器又要改十几个文件的
#   parametrize 栈。耗时是运行期属性不是语义属性, 用可再生成的清单表达更诚实。
#
# 判据: 完整套 profile 里 call 阶段 >= 5.0s。

#

#   pytest -m "" -p no:randomly --durations=0 -o log_cli=false > /tmp/d.txt 2>&1
#   grep -E '^[0-9.]+s call' /tmp/d.txt \
#     | sed -E 's/^([0-9.]+)s call +//;t;d' > /tmp/ids.txt   # 保留整行, ID 可能含空格
#   grep -E '^[0-9.]+s call' /tmp/d.txt \
#     | awk '{split($0,a,"s call "); if ($1+0>=5.0) print a[2]}' | sort > tests/slow_tests.txt
